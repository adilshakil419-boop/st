-- =========================================================
-- 13. CHECK RESULT
-- =========================================================

SELECT
    s.id AS story_id,
    s.title AS story_title,
    a.name AS author,
    c.name AS category,
    ch.chapter_number,
    ch.title AS chapter_title,
    ch.content AS chapter_content
FROM stories s
LEFT JOIN authors a
    ON a.id = s.author_id
LEFT JOIN story_categories sc
    ON sc.story_id = s.id
LEFT JOIN categories c
    ON c.id = sc.category_id
LEFT JOIN chapters ch
    ON ch.story_id = s.id
WHERE s.title = 'কাটা_হাত'
ORDER BY ch.chapter_number;
